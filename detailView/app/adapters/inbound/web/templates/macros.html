{# macros.html — reusable Jinja2 macros imported by all templates. Usage in any template: {% from "macros.html" import badge, caveat, kpi_stat, money_cell %} Macros: badge(label, tone, kind) — Badge chip (text always present; color is not sole signal) caveat(label, severity) — Inline data-quality callout (info / warn) kpi_stat(label, value, sub) — Stat box: label / big-value / optional sub money_cell(amount) — with right-aligned VND or em-dash flag_chips(flags) — Row of caveat chips from list[DataQualityFlag] badge_row(badges) — Row of badge chips from list[Badge] #} {# ------------------------------------------------------------------ #} {# badge — renders a single Badge chip #} {# tone: neutral | good | warn | bad #} {# ------------------------------------------------------------------ #} {% macro badge(label, tone="neutral", kind="") %} {{ label }} {% endmacro %} {# ------------------------------------------------------------------ #} {# badge_row — renders a list of Badge domain objects #} {# ------------------------------------------------------------------ #} {% macro badge_row(badges) %} {% if badges %}
{% for b in badges %} {{ badge(b.label, b.tone.value, b.kind) }} {% endfor %}
{% endif %} {% endmacro %} {# ------------------------------------------------------------------ #} {# caveat — inline data-quality callout #} {# severity: "info" | "warn" #} {# ------------------------------------------------------------------ #} {% macro caveat(label, severity="info") %}
{% if severity == "warn" %}⚠{% else %}ⓘ{% endif %} {{ label }}
{% endmacro %} {# ------------------------------------------------------------------ #} {# flag_chips — renders a list of DataQualityFlag domain objects #} {# ------------------------------------------------------------------ #} {% macro flag_chips(flags) %} {% if flags %}
{% for f in flags %} {{ caveat(f.label, f.severity) }} {% endfor %}
{% endif %} {% endmacro %} {# ------------------------------------------------------------------ #} {# kpi_stat — label / big-value / optional sub text #} {# hero: bool — applies larger value font #} {# ------------------------------------------------------------------ #} {% macro kpi_stat(label, value, sub=None, hero=False) %}
{{ label }} {{ value }} {% if sub is not none and sub != "" %} {{ sub }} {% endif %}
{% endmacro %} {# ------------------------------------------------------------------ #} {# money_cell — with right-aligned VND amount #} {# Expects amount already passed through |vnd filter at call site, #} {# or pass the raw value and let the macro apply it. #} {# ------------------------------------------------------------------ #} {% macro money_cell(amount) %} {{ amount | vnd }} {% endmacro %} {# ------------------------------------------------------------------ #} {# waterfall_row — one row in a revenue waterfall #} {# prefix: "−" / "+" / "=" / "" for visual sign #} {# ------------------------------------------------------------------ #} {% macro waterfall_row(label, amount, prefix="", indent=False, total=False, section=False) %} {% if prefix %}{{ prefix }} {% endif %}{{ label }} {{ amount | vnd }} {% endmacro %}
